cmake_minimum_required(VERSION 3.14)
project(cpp_exercises LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
foreach(example lru_cache dijkstra thread_pool topological_sort bounded_queue text_statistics union_find ring_buffer running_median)
    add_executable(${example} ${example}.cpp)
endforeach()
target_link_libraries(thread_pool PRIVATE Threads::Threads)
target_link_libraries(bounded_queue PRIVATE Threads::Threads)
enable_testing()
add_test(NAME lru_cache COMMAND lru_cache)
add_test(NAME dijkstra COMMAND dijkstra)
add_test(NAME thread_pool COMMAND thread_pool)
add_test(NAME topological_sort COMMAND topological_sort)
add_test(NAME bounded_queue COMMAND bounded_queue)
add_test(NAME union_find COMMAND union_find)
add_test(NAME ring_buffer COMMAND ring_buffer)
add_test(NAME running_median COMMAND running_median)
